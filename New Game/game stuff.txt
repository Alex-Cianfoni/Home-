class NavalWorld:
    """Very small “persistent” world that stores terrain for each theater."""
    @staticmethod
    def generate_initial_theaters() -> List[Dict]:
        return [
            {"name": "Pacific Exercises",   "lat": 30, "lon": -120,
             "terrain": "Island Complex", "base_fuel": 150},
            {"name": "Atlantic Corridor",  "lat": 40, "lon": 5,
             "terrain": "Deep Trench",      "base_fuel": 200},
            {"name": "Mediterranean Base","lat": 35, "lon": -10,
             "terrain": "Coastal Shore",    "base_fuel": 120}
        ]

    @staticmethod
    def get_theater_at(lat: int, lon: int) -> Dict:
        # Nearest theater (simple distance heuristic)
        best = None
        best_dist = math.hypot(1000, 1000)   # huge start value
        for t in NavalWorld.generate_initial_theaters():
            d = math.hypot(lat - t["lat"], lon - t["lon"])
            if d < best_dist:
                best_dist = d
                best = t
        return best

    #  NEW: Persistence helpers                                                 #

    def save_world(self):
        """Write current state to a JSON file."""
        data = {
            "mode": str(self.mode),
            "turn_start": self.turn_start,
            "s_maint": self.ship_maintenance_points,
            "fuel": round(self.player_fuel, 2),
            "supplies": self.supplies.copy(),
            "contacts": [c.__dict__ for c in self.contacts],
            "mission": {
                "id": self.mission.name if self.mission else None,
                "completed": self.mission.completed if self.mission else False,
                "failed":   self.mission.failed  if self.mission else False
            },
            "player_ship": self.player_ship.__dict__ if self.player_ship else None
        }
        with open("rustbucket_world.json", "w") as f:
            json.dump(data, f)

    def load_persistent_state(self):
        """Read JSON file; restore everything we can."""
        try:
            with open("rustbucket_world.json", "r") as f:
                data = json.load(f)
            # Mode
            if "mode" in data and list(MODE_CONFIGS.keys()).index(data["mode"]) == 0:
                self.mode = GameMode(data["mode"])
                self.config = MODE_CONFIGS[self.mode]
            else:
                # keep default mode (Medium) – you could also let player change later
                pass

            # Resources
            self.player_fuel = float(data.get("fuel", 150.0))
            self.supplies = data.get("supplies", [])

            # Maintainments
            self.ship_maintenance_points = int(data["s_maint"])
            self.ship_fuel_consumption_per_km = 2.0   # fixed for now

            # Contacts / Mission state – recreate objects
            if "contacts" in data:
                self.contacts = [
                    Contact(**c) for c in data["contacts"]
                ]
            if "mission" in data and (self.mission is None or not self.mission.completed):
                self.mission = Mission(
                    objective=MissionObjective(data["mission"]["objective"]),
                    description=data.get("description", ""),
                    duration=int(data["duration"]),
                    success_conditions=[s for s in mission.success_conditions if s],
                    failure_conditions=[s for s in mission.failure_conditions if s]
                )
            # Player ship
            if self.player_ship:
                self.player_ship = Ship(**self.player_ship.__dict__)
        except FileNotFoundError:
            pass   # fresh game – start empty

    def get_theater_info(self):
        """Return terrain and base_fuel of the current theater."""

        return {"terrain": t["terrain"], "base_fuel": int(t["base_fuel"]),

    # --------------------------------------------------------------------------- #
    #  NEW: Long‑term crew & ship experience                                   #
    # --------------------------------------------------------------------------- #
    def update_crew_fatigue(self):
        """Fade morale/fatigue based on mission progress."""
        for member in self.player_ship.crew:
            member.fatigue = min(100,
                                 max(0, member.morale + member.expertise * 2 - random.randint(5, 15)))
    def apply_maintenance(self):
        """Use accumulated maintenance points to repair hull."""
        if self.ship_maintenance_points >= 30:
            self.player_ship.hull_integrity = min(100,
                                                 self.player_ship.hull_integrity + 30)
            self.ship_maintenance_points -= 30
    def consume_fuel(self, km: float):
        """Deduct fuel based on distance travelled."""
        self.player_fuel -= km * self.ship_fuel_consumption_per_km

    # --------------------------------------------------------------------------- #
    #  NEW: Mode‑specific scaling (AI etc.)                                    #
    # --------------------------------------------------------------------------- #
    def _apply_difficulty_scaling(self):
        """Called each turn – uses the current mode config."""
        for contact in self.contacts:
            if random.random() < self.config.detection_chance and \
               contact.detection_state == DetectionState.UNDETECTED:
                # 1️⃣ Suspect
                contact.detection_state = DetectionState.SUSPECTED
                self.message_log.append(f"Contact {contact.name} suspected!")

            elif random.random() < self.config.detection_confidence and \
                 contact.detection_state == DetectionState.SUSPECTED:
                # 2️⃣ Locate
                contact.detection_state = DetectionState.LOCATED

        # Stealth decay per detected enemy
        loss_per_contact = max(0, self.config.stealth_loss_per_contact)
        for c in self.contacts:
            if c.detection_state != DetectionState.UNDETECTED and \
               c.contact_type == ContactType.SUBMARINE:
                self.player_ship.stealth_level = max(
                    0,
                    self.player_ship.stealth_level - loss_per_contact)

        # Hull damage per enemy contact (mode‑dependent)
        if any(c.detection_state in [DetectionState.LOCATED, DetectionState.TRACKED]
               for c in self.contacts):
            extra_damage = int(self.config.hull_damage_multiplier * 15)
            self.player_ship.hull_integrity = max(
                0,
                self.player_ship.hull_integrity - extra_damage)

    # --------------------------------------------------------------------------- #
    #  NEW: Resource management                                                #
    # --------------------------------------------------------------------------- #
    def refill_from_depot(self, depot_name: str):
        """Find a depot with enough supplies and deduct fuel."""
        for d in self.supplies:
            if d["name"] == depot_name:
                self.player_fuel = min(200.0,
                                       self.player_fuel + 50)   # depot gives 50 L
                self.message_log.append(f"Refueled at {depot_name}.")
                break

    def update_suggested_actions(self):
        """Add mission‑specific actions that depend on resources / hull."""
        if self.player_ship.hull_integrity < 30:
            self.suggested_actions.append("Repair hull (maintenance)")

        # If fuel low, suggest refueling
        if self.player_fuel < 20:
            self.suggested_actions.append("Refuel from depot")
            # Look for nearest depot (simple distance)
            d = NavalWorld.get_theater_at(self.player_ship.position[0],
                                          self.player_ship.position[1])
            if "island" in d["terrain"] and self.player_fuel < 5:
                self.refill_from_depot(d["name"])
        else:
            # Normal actions
            normal = [
                "Change depth",
                "Adjust speed",
                "Monitor contacts",
                "Communicate with allies"
            ]
            if len(self.suggested_actions) > 4:   # avoid duplicates
                self.suggested_actions[:4] = normal

        self.screen.blit(title,
                         (SCREEN_WIDTH//2 - title.get_width()//2, 100))

        score_txt = self.font_medium.render(f"Final Score: {self.score}/100",
                                          True, WHITE)
        self.screen.blit(score_txt,
                         (SCREEN_WIDTH//2 - score_txt.get_width()//2, 200))

        if self.mission.completed:
            res = self.font_medium.render("Mission Successful!", True, GREEN)
        else:
            res = self.font_medium.render("Mission Failed", True, RED)

        self.screen.blit(res,
                         (SCREEN_WIDTH//2 - res.get_width()//2, 250))

        # Mission results
        pygame.draw.rect(self.screen, LIGHT_BLUE,
                         (220, 300, 800, 300), 2)
        py = 320
        results = [
            f"Primary Objective: {'✓' if self.mission.completed else '✗'}",
            f"Extraction: {'✓' if self.mission.completed else '✗'}",
            f"Confirmed Detections: {sum(1 for c in self.contacts
                                          if c.detection_state != DetectionState.UNDETECTED)}",

            f"Battery Remaining: {int(self.player_fuel)} L",
            f"Torpedoes Left: {len(self.player_ship.weapons)}"

            txt = self.font_small.render(r, True, WHITE)
            self.screen.blit(txt,
                             (230, py))
            py += 35

        restart = self.font_medium.render("Press R to Restart", True, GREEN)
        self.screen.blit(restart,
                         (SCREEN_WIDTH//2 - restart.get_width()//2, 670))

        pygame.display.flip()

    # --------------------------------------------------------------------------- #
    #  Core turn processing                                                      #
    # --------------------------------------------------------------------------- #
    def process_command(self, command: str):
        """Handle player input and drive the turn."""
        cmd = command.lower().strip()
        self.message_log.append(f"Command: {cmd}")

        if "depth" in cmd:
            new_d = random.randint(30, 80)
            self.player_ship.depth = new_d
            self.message_log.append(f"Depth changed to {new_d} m")

        elif "speed" in cmd:
            # speed is limited by fuel consumption
            max_speed = min(25,
                            int(self.player_fuel * 0.4))   # rough limit
            new_s = random.randint(5, max_speed)
            self.player_ship.speed = new_s
            self.message_log.append(f"Speed set to {new_s} knots")

        elif "sonar" in cmd or "contact" in cmd:
            self.message_log.append("Sonar contacts monitored")
            # Simulate detection (already covered by _apply_difficulty_scaling)

        elif "communicate" in cmd or "radio" in cmd:
            self.message_log.append("Communication established with exercise command")

        elif "stealth" in cmd:
            inc = random.randint(0, 12)
            self.player_ship.stealth_level = min(100,
                                                self.player_ship.stealth_level + inc)
            self.message_log.append(f"Stealth increased to {self.player_ship.stealth_level}%")

        else:
            self.message_log.append("Command processed – proceeding with standard ops.")

        # Apply difficulty scaling
        self._apply_difficulty_scaling()

        # Consume fuel (distance = 1 km per turn for simplicity)
        self.consume_fuel(1)

        # Update long‑term crew/fatigue
        self.update_crew_fatigue()

        # Maintenance points accrual if idle (ship speed < 5)
        if self.player_ship.speed < 5:
            self.ship_maintenance_points = min(
                200,
                self.ship_maintenance_points + self.ship_maintenance
            )
            # Use them immediately for hull repair?
            if self.ship_maintenance_points >= 30:
                self.apply_maintenance()

        # Mission check
        self.check_mission_conditions()

        self.turn += 1

    def simulate_opponent_reactions(self):
        """AI reacts according to the current mode."""
        self._apply_difficulty_scaling()   # already does detection & damage

        # Example: occasional extra damage from environment (realistic only)
        if self.mode == GameMode.REALISTIC and random.random() < 0.15:
            for c in self.contacts:
                if c.detection_state == DetectionState.LOCATED:
                    self.player_ship.hull_integrity = max(
                        0,
                        self.player_ship.hull_integrity - 20)

    def check_mission_conditions(self):
        """End‑of‑turn logic."""
        # Time out
        if self.turn >= self.mission.duration:
            self.score += self.config.mission_success_bonus
            self.message_log.append("Mission deadline reached.")
            return

        # Hull broken?
        if self.player_ship.hull_integrity < 30:
            self.ship_maintenance_points = 0   # no repair possible
            self.message_log.append(f"Hull integrity below threshold – mission aborted!")

        # Battery dead?
        if self.player_fuel <= 0:
            self.message_log.append("Battery depleted – cannot continue.")

    def update_suggested_actions(self):
        """Refresh the list of actions for UI."""
        self.suggested_actions = [
            "Change depth",
            "Adjust speed",
            "Monitor contacts",
            "Communicate with allies"
        ]

        # Add mode‑specific actions (already in _apply_difficulty_scaling)
        if self.player_ship.hull_integrity < 30:
            self.suggested_actions.append("Repair hull (maintenance)")

        if self.player_fuel < 20:
            d = NavalWorld.get_theater_at(self.player_ship.position[0],
                                          self.player_ship.position[1])
            if "island" in d["terrain"]:
                self.suggested_actions.append(f"Refuel at {d['name']}")

    # --------------------------------------------------------------------------- #
    #  Main loop                                                               #
    # --------------------------------------------------------------------------- #
    def run(self):
        while True:
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    return
                if self.state == GameState.MENU and event.type == pygame.KEYDOWN \
                        and event.key == pygame.K_SPACE:
                    self.state = GameState.MISSION
                    # Reset turn start for mission length
                    self.turn_start = self.turn
                    break

            if self.state == GameState.MISSION:
                self.process_command("")   # placeholder – real input in draw_menu later
                self.update_suggested_actions()
                self.draw_mission_screen()
            elif self.state == GameState.ENDGAME:
                self.draw_endgame()

            self.clock.tick(FPS)
